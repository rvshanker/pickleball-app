function generateBalancedSchedule() { // 1. Get current settings const numPlayers = parseInt(document.getElementById('playerCount').value); const numCourts = parseInt(document.getElementById('courtCount').value); const numRounds = parseInt(document.getElementById('roundCount').value); // 2. Prepare Players and Tally let players = Array.from({length: numPlayers}, (_, i) => `P${i + 1}`); const gameTracker = {}; players.forEach(p => gameTracker[p] = 0); // 3. Circle Rotation Logic (Fixed Pivot) // If odd, we add a 'GHOST' player so the circle is even. // Anyone paired with the GHOST is actually "Sitting Out". let circle = [...players]; if (circle.length % 2 !== 0) circle.push("GHOST"); const n = circle.length; const fullSchedule = []; for (let r = 0; r < numRounds; r++) { const roundPairs = []; const waitingList = []; // Pair from outside-in (0-15, 1-14, 2-13, etc.) for (let i = 0; i < n / 2; i++) { const p1 = circle[i]; const p2 = circle[n - 1 - i]; if (p1 === "GHOST") waitingList.push(p2); else if (p2 === "GHOST") waitingList.push(p1); else roundPairs.push([p1, p2]); } // 4. Assign Matches to Courts (2 pairs per court for doubles) const matches = []; for (let i = 0; i < roundPairs.length; i += 2) { if (matches.length < numCourts && (i + 1) < roundPairs.length) { const m = { team1: roundPairs[i], team2: roundPairs[i+1] }; matches.push(m); // Track games played for stats m.team1.concat(m.team2).forEach(player => gameTracker[player]++); } else { // Not enough courts or leftover pair? They sit out this round. waitingList.push(...roundPairs[i]); if (roundPairs[i+1]) waitingList.push(...roundPairs[i+1]); } } fullSchedule.push({ round: r + 1, matches, waiting: waitingList }); // 5. THE KEY: Rotate the circle (keep first player fixed at index 0) const fixed = circle[0]; const rotators = circle.slice(1); rotators.unshift(rotators.pop()); // Shift last element to the front circle = [fixed, ...rotators]; } renderTournament(fullSchedule, gameTracker); }
